def headers(path):
    txt=path.read_text()
    out={}
    for m in re.finditer(r'^theorem\s+(\w+)\s+',txt,re.M):
        finish=txt.index(':=',m.end())
        out[m.group(1)]=' '.join(txt[m.end():finish].split())
    return out
