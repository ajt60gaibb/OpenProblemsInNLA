def polymul(p,q):
    out=[Q(0)]*(len(p)+len(q)-1)
    for i,x in enumerate(p):
        for j,y in enumerate(q):out[i+j]+=x*y
    return out

def ldl_positive(matrix):
    d=[row[:] for row in matrix];pivots=[]

    return pivots
