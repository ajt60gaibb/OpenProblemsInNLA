from pathlib import Path
import datetime, hashlib, json, re, subprocess
P=Path(__file__).resolve().parents[2];E=Path(__file__).resolve().parent;W=P.parents[2]
C=Path('/tmp/nla-lean-mi22-worktree/matrix-inequalities-and-norms/MI-22/lean/.lake/packages')
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def git(*args):return subprocess.check_output(['git','-C',str(W),*args])
pf=P/'verification/proof-freeze.json';sf=P/'reviews/statement-freeze.json'
assert sha(pf)=='ab05e2bf801e6906453e88a4d84d5e73191f776db05610fb8b8e5e4a03d4f856'
assert sha(P/'reviews/proof-completion.md')=='dab07a0cf99d739914c54efea282211afcbfb45507ac3fef74b285afc3ad79ce'
f=json.loads(pf.read_text());s=json.loads(sf.read_text())
assert len(f['files'])==450 and len(s['files'])==39
for n,h in f['files'].items():assert sha(P/n)==h,n
for n,h in s['files'].items():assert sha(P/n)==h,n
assert git('rev-parse','HEAD').decode().strip()==f['base']
assert not git('diff','--name-only') and not git('diff','--cached','--name-only')
originals={}
for n,h in f['source_files'].items():
    raw=git('show',f['base']+':'+n)
    assert hashlib.sha256(raw).hexdigest()==h and (W/n).read_bytes()==raw
    blob=git('rev-parse',f['base']+':'+n).decode().strip()
    assert blob==f['source_git_blobs'][n]
    originals[n]={'sha256':h,'git_blob':blob,'unchanged':True}
assert '**Status:** Solved' in (W/'randomized-and-low-rank-approximation/RA-08/README.md').read_text()
challenge=(P/'Challenge.lean').read_text();solution=(P/'Solution.lean').read_text()
def sigs(text):
    return {name:' '.join(body.split()) for name,body in
            re.findall(r'^theorem\s+(\w+)(.*?)\s*:=\s*by\b',text,re.S|re.M)}
assert sigs(challenge)==sigs(solution) and len(sigs(solution))==14
cfg=json.loads((P/'comparator.json').read_text())
assert cfg=={'challenge_module':'Challenge','solution_module':'Solution',
 'theorem_names':['NLA.RA08.'+n for n in sigs(challenge)],'definition_names':[],
 'permitted_axioms':['propext','Classical.choice','Quot.sound']}
math_sources=[*sorted((P/'NLA/RA08').glob('*.lean')),P/'Solution.lean']
imports={}
for path in math_sources:
    code=re.sub(r'/\-.*?\-/','',path.read_text(),flags=re.S)
    code=re.sub(r'--[^\n]*','',code)
    assert not re.search(r'\b(sorry|admit|native_decide|unsafe|axiom)\b',code),path
    ims=re.findall(r'^import\s+(\S+)',code,re.M);imports[str(path.relative_to(P))]=ims
    assert 'Challenge' not in ims and all(not i.startswith('reviews.') for i in ims)

assert result['verdict']=='PASS' and len(result['commands'])==20
assert result['pins_rechecked_after'] and result['frozen_proof_inputs_preserved']==450
assert result['disposable_prefix_removed_after_check'] and not Path(result['prefix']).exists()
axioms=[]
for row in result['commands']:
    assert row['exit_code']==0 and sha(P/row['source'])==row['source_sha256']
    assert sha(run/row['log'])==row['log_sha256']
    text=(run/row['log']).read_text()
    if row['source']=='Challenge.lean':assert text.count('warning:')==text.count('declaration uses `sorry`')==14
    else:assert 'warning:' not in text and 'error:' not in text
    for name,values in re.findall(r"'([^'\n]+)' depends on axioms: \[([^\]]*)\]",text):
        ax=[a.strip() for a in values.split(',')]
        assert set(ax)==set(cfg['permitted_axioms'])
        axioms.append({'declaration':name,'axioms':ax,'log':row['log']})
assert len(axioms)==61
