open NLA.Statements.RowDeletion
example (θ a : ℝ)
    (hθ0 : 0 < θ) (hθ1 : θ < 1) (ha : 0 < a)
    (hquantile : GaussianQuantile θ a) :
    ∀ ε : ℝ, 0 < ε → ∃ N : ℕ,
      ∀ n m : ℕ, 0 < n → 0 < m → N ≤ n →
        ∀ A : Matrix (Fin m) (Fin n) ℝ, UnitRows A →
          (n : ℝ) / (m : ℝ) * SubsingularSquared θ A ≤ TrimmedMoment a + ε := NLA.Proofs.IE22.uniform_squared_upper θ a hθ0 hθ1 ha hquantile
example (θ a : ℝ)
    (hθ0 : 0 < θ) (hθ1 : θ < 1) (ha : 0 < a)
    (hquantile : GaussianQuantile θ a) :
    ∀ ε : ℝ, 0 < ε → ∀ N R : ℕ,
      ∃ n m : ℕ, ∃ A : Matrix (Fin m) (Fin n) ℝ,
        0 < n ∧ 0 < m ∧ N ≤ n ∧ R * n ≤ m ∧ UnitRows A ∧
          TrimmedMoment a - ε ≤ (n : ℝ) / (m : ℝ) * SubsingularSquared θ A := NLA.Proofs.IE22.near_sharp_unit_row_witness θ a hθ0 hθ1 ha hquantile
