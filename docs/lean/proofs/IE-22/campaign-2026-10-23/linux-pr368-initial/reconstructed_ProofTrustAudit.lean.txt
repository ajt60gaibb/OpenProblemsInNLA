import Solution
import LeanCert.Tactic.Verification
set_option autoImplicit false
set_option leancert.trust "kernel"

open Lean Elab Command in
elab "#assert_proof_constant " n:ident : command => do
  let decl ← liftCoreM <| realizeGlobalConstNoOverloadWithInfo n
  match (← getEnv).find? decl with
  | some (.thmInfo _) => pure ()
  | _ => throwErrorAt n "expected a proved theorem constant"

#assert_proof_constant NLA.Proofs.IE22.uniform_squared_upper
#assert_trust kernel NLA.Proofs.IE22.uniform_squared_upper
#print axioms NLA.Proofs.IE22.uniform_squared_upper
#assert_proof_constant NLA.Proofs.IE22.near_sharp_unit_row_witness
#assert_trust kernel NLA.Proofs.IE22.near_sharp_unit_row_witness
#print axioms NLA.Proofs.IE22.near_sharp_unit_row_witness
#assert_proof_constant NLA.Proofs.IE22.target
#assert_trust kernel NLA.Proofs.IE22.target
