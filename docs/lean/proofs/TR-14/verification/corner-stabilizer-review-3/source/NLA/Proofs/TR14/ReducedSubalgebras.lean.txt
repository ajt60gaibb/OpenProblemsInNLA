/-! The reduced-subalgebra finiteness step in TR-14's contextual product
argument. Only reduced subalgebras are counted; ambient nilpotents are allowed. -/
set_option autoImplicit false
set_option leancert.trust "kernel"

namespace NLA.Proofs.TR14

open scoped BigOperators
noncomputable section

/-- A reduced finite-dimensional commutative complex algebra is spanned
by its actual idempotents. The residue fields in the Artinian decomposition
are proved to be scalar copies of the complex numbers. -/
theorem reduced_algebra_span_idempotents (B : Type*) [CommRing B]
    [Algebra ℂ B] [FiniteDimensional ℂ B] [IsReduced B] :
    Submodule.span ℂ {e : B | e * e = e} = ⊤ := by
  classical
  let : IsArtinianRing B := isArtinian_of_tower ℂ inferInstance
  let ι := MaximalSpectrum B
  let : Fintype ι := Fintype.ofFinite ι
  let φ : B ≃ₐ[ℂ] (∀ m : ι, B ⧸ m.asIdeal) :=
    (IsArtinianRing.equivPi B).restrictScalars ℂ
  have hsurj (m : ι) : Function.Surjective (algebraMap ℂ (B ⧸ m.asIdeal)) := by
    let : m.asIdeal.IsMaximal := m.isMaximal
    let : Field (B ⧸ m.asIdeal) := Ideal.Quotient.field m.asIdeal
    let : Module.Finite ℂ (B ⧸ m.asIdeal) :=
      Module.Finite.of_surjective (Ideal.Quotient.mkₐ ℂ m.asIdeal).toLinearMap
        (Ideal.Quotient.mkₐ_surjective ℂ m.asIdeal)
    exact (IsAlgClosed.algebraMap_bijective_of_isIntegral
      (k := ℂ) (K := B ⧸ m.asIdeal)).2
  let e : ι → B := fun m => φ.symm (Pi.single m 1)
  have he (m : ι) : e m * e m = e m := by
    apply φ.injective
    simp only [map_mul, e, AlgEquiv.apply_symm_apply]
    funext j
    by_cases h : m = j
    · subst j
      simp
    · simp [h]
  apply top_unique
  intro x _hx
  choose c hc using fun m : ι => hsurj m (φ x m)
  have hxe : x = ∑ m : ι, c m • e m := by
    apply φ.injective
    rw [map_sum]
    funext j
    simp only [map_smul, e, AlgEquiv.apply_symm_apply, Finset.sum_apply, Pi.smul_apply]
    simp only [Algebra.smul_def]
    rw [Finset.sum_eq_single j]
    · simpa only [Pi.single_eq_same, mul_one] using (hc j).symm
    · intro m _hm hne
      rw [Pi.single_eq_of_ne' hne, mul_zero]
    · intro hj
      exact False.elim (hj (Finset.mem_univ j))
  rw [hxe]
  exact Submodule.sum_mem _ (fun m _ =>
    Submodule.smul_mem _ _ (Submodule.subset_span (he m)))

variable {B : Type*} [CommRing B] [Algebra ℂ B] [FiniteDimensional ℂ B]

/-- The ambient linear subspace of a reduced subalgebra is generated by
exactly its own ambient idempotents. -/
theorem reducedSubalgebra_eq_span_idempotents (S : Subalgebra ℂ B) [IsReduced S] :
    S.toSubmodule = Submodule.span ℂ {e : B | e ∈ S ∧ e * e = e} := by
  apply le_antisymm
  · intro x hx
    have hmem : (⟨x, hx⟩ : S) ∈ Submodule.span ℂ {e : S | e * e = e} := by
      rw [reduced_algebra_span_idempotents S]
      exact Submodule.mem_top
    have hmap : x ∈ (Submodule.span ℂ {e : S | e * e = e}).map S.val.toLinearMap :=
      Submodule.mem_map.mpr ⟨⟨x, hx⟩, hmem, rfl⟩
    rw [Submodule.map_span] at hmap
    apply Submodule.span_mono (show
      S.val '' {e : S | e * e = e} ⊆ {e : B | e ∈ S ∧ e * e = e} from ?_) hmap
    rintro _ ⟨e, he, rfl⟩
    exact ⟨e.property, congrArg (fun z : S => (z : B)) he⟩
  · exact Submodule.span_le.mpr (fun _ he => he.1)

/-- Finite ambient idempotents imply finitely many reduced subalgebras.
No finiteness claim about arbitrary nonreduced subalgebras is made. -/
theorem finite_reducedSubalgebras
    (hfin : {e : B | e * e = e}.Finite) :
    {S : Subalgebra ℂ B | IsReduced S}.Finite := by
  classical
  let I := {e : B // e * e = e}
  let : Finite I := hfin.to_subtype
  let f : {S : Subalgebra ℂ B // IsReduced S} → Set I :=
    fun S => {e | (e : B) ∈ S.val}
  have hf : Function.Injective f := by
    rintro ⟨S, hS⟩ ⟨T, hT⟩ hST
    apply Subtype.ext
    let : IsReduced S := hS
    let : IsReduced T := hT
    apply Subalgebra.toSubmodule_injective
    rw [reducedSubalgebra_eq_span_idempotents S, reducedSubalgebra_eq_span_idempotents T]
    congr 1
    ext x
    constructor
    · rintro ⟨hx, hi⟩
      refine ⟨?_, hi⟩
      have heq := congrArg (fun U : Set I => (⟨x, hi⟩ : I) ∈ U) hST
      exact (Eq.mp heq hx)
    · rintro ⟨hx, hi⟩
      refine ⟨?_, hi⟩
      have heq := congrArg (fun U : Set I => (⟨x, hi⟩ : I) ∈ U) hST
      exact (Eq.mpr heq hx)
  exact Set.finite_coe_iff.mp (Finite.of_injective f hf)

/-- A commutative Artinian ring has finitely many actual idempotents,
including when the ring itself has nilpotents. -/
theorem finite_idempotents_of_artinian (R : Type*) [CommRing R] [IsArtinianRing R] :
    {e : R | e * e = e}.Finite := by
  have hclopen : Finite (TopologicalSpace.Clopens (PrimeSpectrum R)) :=
    Finite.of_injective (fun U : TopologicalSpace.Clopens (PrimeSpectrum R) =>
      (U : Set (PrimeSpectrum R))) SetLike.coe_injective
  let : Finite (TopologicalSpace.Clopens (PrimeSpectrum R)) := hclopen
  have hfinite : Finite {e : R // IsIdempotentElem e} :=
    Finite.of_injective (PrimeSpectrum.isIdempotentElemEquivClopens (R := R))
      (PrimeSpectrum.isIdempotentElemEquivClopens (R := R)).injective
  exact Set.finite_coe_iff.mp hfinite

/-- The actual finiteness needed in the contextual product argument holds
for every finite-dimensional commutative complex ambient algebra. -/
theorem finite_reducedSubalgebras_of_finiteDimensional :
    {S : Subalgebra ℂ B | IsReduced S}.Finite := by
  let : IsArtinianRing B := isArtinian_of_tower ℂ inferInstance
  exact finite_reducedSubalgebras (finite_idempotents_of_artinian B)

#assert_trust kernel reduced_algebra_span_idempotents
#assert_trust kernel reducedSubalgebra_eq_span_idempotents
#assert_trust kernel finite_reducedSubalgebras
#assert_trust kernel finite_idempotents_of_artinian
#assert_trust kernel finite_reducedSubalgebras_of_finiteDimensional
#print axioms finite_reducedSubalgebras_of_finiteDimensional

end
