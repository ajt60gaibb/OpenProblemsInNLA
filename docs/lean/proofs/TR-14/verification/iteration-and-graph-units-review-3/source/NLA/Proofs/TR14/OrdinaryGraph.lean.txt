/-! Actual graph spaces of arbitrary ordinary mode factors, TR-14 Section 5.
The quotient-mode map is a parameter so the same exact normalized apolar
chart can be retained. No nonzero factor-entry premise is used. -/

/-- The actual dual mode functional of an ordinary factor. -/
def ordinaryModeForm {m n r : ℕ} (u : Fin r → Fin m → Fin n → ℂ)
    (j : Fin r) (k : Fin m) : (Fin n → ℂ) →ₗ[ℂ] ℂ where
  toFun p := ∑ i, u j k i * p i
  map_add' p q := by simp [mul_add, Finset.sum_add_distrib]
  map_smul' c p := by simp [Finset.mul_sum, mul_left_comm]

@[simp] theorem ordinaryModeForm_apply {m n r : ℕ}
    (u : Fin r → Fin m → Fin n → ℂ) (j : Fin r) (k : Fin m) (p : Fin n → ℂ) :
    ordinaryModeForm u j k p = ∑ i, u j k i * p i := rfl

@[simp] theorem ordinaryModeForm_single {m n r : ℕ}
    (u : Fin r → Fin m → Fin n → ℂ) (j : Fin r) (k : Fin m) (i : Fin n) :
    ordinaryModeForm u j k (Pi.single i 1) = u j k i := by
  classical
  simp [ordinaryModeForm, Pi.single_apply]

/-- Minimum-summand independence gives nonzero dual mode functionals. -/
theorem ordinaryModeForm_ne_zero {m n r : ℕ}
    (u : Fin r → Fin m → Fin n → ℂ)
    (hind : LinearIndependent ℂ (ordinarySummand u)) (j : Fin r) (k : Fin m) :
    ordinaryModeForm u j k ≠ 0 := by
  intro hz
  apply ordinary_mode_ne_zero_of_independent u hind j k
  funext i
  have h := DFunLike.congr_fun hz (Pi.single i 1)
  simpa only [ordinaryModeForm_single, LinearMap.zero_apply, Pi.zero_apply] using h

variable {A : Type*} [CommRing A] [Algebra ℂ A]

/-- Mode vectors map to their actual quotient component and all actual
ordinary-factor functional values. -/
def ordinaryGraphMap {m n r : ℕ} (φ : (Fin n → ℂ) →ₗ[ℂ] A)
    (u : Fin r → Fin m → Fin n → ℂ) (k : Fin m) :
    (Fin n → ℂ) →ₗ[ℂ] (A × (Fin r → ℂ)) where
  toFun p := (φ p, fun j => ordinaryModeForm u j k p)
  map_add' p q := by ext j <;> simp [mul_add, Finset.sum_add_distrib]
  map_smul' c p := by ext j <;> simp

@[simp] theorem ordinaryGraphMap_apply {m n r : ℕ} (φ : (Fin n → ℂ) →ₗ[ℂ] A)
    (u : Fin r → Fin m → Fin n → ℂ) (k : Fin m) (p : Fin n → ℂ) :
    ordinaryGraphMap φ u k p = (φ p, fun j => ordinaryModeForm u j k p) := rfl

def ordinaryGraphSpace {m n r : ℕ} (φ : (Fin n → ℂ) →ₗ[ℂ] A)
    (u : Fin r → Fin m → Fin n → ℂ) (k : Fin m) : Submodule ℂ (A × (Fin r → ℂ)) :=
  LinearMap.range (ordinaryGraphMap φ u k)

/-- Exact first projection, used by the all-but-one quotient product bridge. -/
theorem ordinaryGraphSpace_fst {m n r : ℕ} (φ : (Fin n → ℂ) →ₗ[ℂ] A)
    (u : Fin r → Fin m → Fin n → ℂ) (k : Fin m) :
    (ordinaryGraphSpace φ u k).map (AlgHom.fst ℂ A (Fin r → ℂ)).toLinearMap =
      LinearMap.range φ := by
  ext x
  constructor
  · rintro ⟨y, ⟨p, rfl⟩, rfl⟩
    exact ⟨p, rfl⟩
  · rintro ⟨p, rfl⟩
    exact ⟨ordinaryGraphMap φ u k p, ⟨p, rfl⟩, rfl⟩

/-- Finite products of actual linear ranges are generated by their pure
product images. This is an exact span equality. -/
theorem product_ranges_eq_span {ι M V : Type*} [Fintype ι] [AddCommGroup V]
    [Module ℂ V] [CommRing M] [Algebra ℂ M] (L : ι → V →ₗ[ℂ] M) :
    (∏ k, LinearMap.range (L k)) =
      Submodule.span ℂ (Set.range (fun p : ι → V => ∏ k, L k (p k))) := by
  classical
  have hr (k : ι) : LinearMap.range (L k) = Submodule.span ℂ (Set.range (L k)) :=
    (Submodule.span_eq (LinearMap.range (L k))).symm
  simp_rw [hr]
  rw [Submodule.prod_span]
  congr 1
  ext z
  rw [Set.mem_fintype_prod]
  constructor
  · rintro ⟨x, hx, hprod⟩
    choose p hp using hx
    exact ⟨p, by simpa only [hp] using hprod⟩
  · rintro ⟨p, rfl⟩
    exact ⟨fun k => L k (p k), fun k => ⟨p k, rfl⟩, rfl⟩

/-- The actual decomposition scalar coordinates span their whole scalar
space when its rank-one coordinate functions are independent. -/
theorem ordinary_evaluations_span_top {m n r : ℕ}
    (u : Fin r → Fin m → Fin n → ℂ)
    (hind : LinearIndependent ℂ (ordinarySummand u)) :
    Submodule.span ℂ (Set.range (fun i : Fin m → Fin n => fun j => ordinarySummand u j i)) = ⊤ := by
  classical
  let M : Matrix (Fin r) (Fin m → Fin n) ℂ := ordinarySummand u
  have hM : LinearIndependent ℂ M.row := hind
  have hr : M.rank = r := by simpa using hM.rank_matrix
  rw [Matrix.rank_eq_finrank_span_cols] at hr
  have hcol : M.col = (fun i j => ordinarySummand u j i) := by
    funext i j
    rfl
  rw [hcol] at hr
  apply Submodule.eq_top_of_finrank_eq
  simpa using hr

/-- The product of the actual graph spaces surjects onto all ordinary
summand scalar coordinates. -/
theorem ordinaryGraph_product_snd {m n r : ℕ} (φ : (Fin n → ℂ) →ₗ[ℂ] A)
    (u : Fin r → Fin m → Fin n → ℂ)
    (hind : LinearIndependent ℂ (ordinarySummand u)) :
    (∏ k, ordinaryGraphSpace φ u k).map (AlgHom.snd ℂ A (Fin r → ℂ)).toLinearMap = ⊤ := by
  classical
  have hspan := ordinary_evaluations_span_top u hind
  apply top_unique
  rw [← hspan]
  apply Submodule.span_le.mpr
  rintro _ ⟨i, rfl⟩
  apply Submodule.mem_map.mpr
  refine ⟨∏ k, ordinaryGraphMap φ u k (Pi.single (i k) 1), ?_, ?_⟩
  · rw [show (∏ k, ordinaryGraphSpace φ u k) =
      Submodule.span ℂ (Set.range (fun p : Fin m → (Fin n → ℂ) =>
        ∏ k, ordinaryGraphMap φ u k (p k))) from product_ranges_eq_span _]
    exact Submodule.subset_span ⟨fun k => Pi.single (i k) 1, rfl⟩
  · funext j
    simp only [AlgHom.toLinearMap_apply, AlgHom.snd_apply, Finset.prod_apply,
      Prod.snd_prod, ordinaryGraphMap_apply, ordinaryModeForm_single, ordinarySummand]

/-- The ordinary decomposition is the actual multilinear identity, obtained
by finite expansion, with no restriction on factors. -/
theorem ordinary_multilinear_identity {m n r : ℕ}
    (u : Fin r → Fin m → Fin n → ℂ) (H : (Fin m → Fin n) → ℂ)
    (hH : ∀ i, H i = ∑ j, ordinarySummand u j i)
    (p : Fin m → Fin n → ℂ) :
    (∑ i, H i * ∏ k, p k (i k)) = ∑ j, ∏ k, ordinaryModeForm u j k (p k) := by
  simp_rw [hH, Finset.sum_mul]
  rw [Finset.sum_comm]
  apply Finset.sum_congr rfl
  intro j _
  change (∑ i : Fin m → Fin n, (∏ k : Fin m, u j k (i k)) * ∏ k : Fin m, p k (i k)) =
    ∏ k : Fin m, ∑ i : Fin n, u j k i * p k i
  rw [Fintype.prod_sum]
  apply Finset.sum_congr rfl
  intro i _
  exact (Finset.prod_mul_distrib).symm

/-- The functional used by the source, with actual scalar coordinates. -/
def ordinaryGraphFunctional {r : ℕ} (Λ : A →ₗ[ℂ] ℂ) :
    (A × (Fin r → ℂ)) →ₗ[ℂ] ℂ where
  toFun x := Λ x.1 - ∑ j, x.2 j
  map_add' x y := by simp [Finset.sum_add_distrib]; ring
  map_smul' c x := by
    change Λ (c • x.1) - ∑ j, c * x.2 j = c * (Λ x.1 - ∑ j, x.2 j)
    rw [map_smul, smul_eq_mul, mul_sub, Finset.mul_sum]

/-- Exact annihilation of the actual product graph space from the quotient
moment identity and actual ordinary decomposition. -/
theorem ordinaryGraph_product_annihilated {m n r : ℕ}
    (φ : (Fin n → ℂ) →ₗ[ℂ] A) (Λ : A →ₗ[ℂ] ℂ)
    (u : Fin r → Fin m → Fin n → ℂ) (H : (Fin m → Fin n) → ℂ)
    (hH : ∀ i, H i = ∑ j, ordinarySummand u j i)
    (hmoment : ∀ p : Fin m → Fin n → ℂ,
      Λ (∏ k, φ (p k)) = ∑ i, H i * ∏ k, p k (i k)) :
    (∏ k, ordinaryGraphSpace φ u k) ≤ LinearMap.ker (ordinaryGraphFunctional Λ) := by
  classical
  rw [show (∏ k, ordinaryGraphSpace φ u k) =
      Submodule.span ℂ (Set.range (fun p : Fin m → (Fin n → ℂ) =>
        ∏ k, ordinaryGraphMap φ u k (p k))) from product_ranges_eq_span _]
  apply Submodule.span_le.mpr
  rintro _ ⟨p, rfl⟩
  change Λ ((∏ k, ordinaryGraphMap φ u k (p k)).1) -
    ∑ j, (∏ k, ordinaryGraphMap φ u k (p k)).2 j = 0
  simp only [Prod.fst_prod, Prod.snd_prod, Finset.prod_apply, ordinaryGraphMap_apply]
  rw [hmoment p, ordinary_multilinear_identity u H hH p, sub_self]

#assert_trust kernel ordinaryGraphSpace_fst
#assert_trust kernel ordinaryGraph_product_snd
#assert_trust kernel ordinaryGraph_product_annihilated
#print axioms ordinaryGraph_product_annihilated

end
