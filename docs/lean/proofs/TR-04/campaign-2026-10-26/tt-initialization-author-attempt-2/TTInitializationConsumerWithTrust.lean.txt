-- Composed original-input endpoint: actual first SVD readiness entails the
-- actual allocated U/sigma/V pointers and retained original dense input.
example (S : Shape) (r : Ranks S) (A : Tensor S) (trace : Trace ttProgram)
    (ht : LegalTrace ttProgram (Start ttProgram S r A) trace) :
    ∃ time : ℕ,time≤102*(Size S r+1) ∧
      (trace.state time).label=167 ∧ (trace.state time).status=.running ∧
      (trace.state time).memory=InputMemory S r A ∧
      (trace.state time).natural 72=(ofShape S).firstU ∧
      (trace.state time).natural 73=(ofShape S).firstSigma ∧
      (trace.state time).natural 74=(ofShape S).firstV ∧
      (trace.state time).natural 76=0 ∧
      CostThrough ttProgram trace time≤102*(Size S r+1) := by
  obtain ⟨n,_,h,htime,hcost⟩ := ttInitial_svd_ready S r A trace ht
  exact ⟨ttFirstRankEntryTime S r+n,htime,h.label,h.status,h.memory,
    h.allocation.firstU,h.allocation.firstSigma,h.allocation.firstV,h.candidateZero,hcost⟩

-- Requested rank zero is still read literally; this support theorem does not
-- silently narrow the header domain to consistent or positive ranks.
example (S : Shape) (A : Tensor S) (trace : Trace ttProgram)
    (ht : LegalTrace ttProgram (Start ttProgram S (fun _ => 0) A) trace) :
    ∃ time : ℕ,(trace.state time).label=167 ∧ (trace.state time).natural 80=0 ∧
      (trace.state time).natural 106=0 := by
  obtain ⟨n,_,h,_,_⟩ := ttInitial_svd_ready S (fun _ => 0) A trace ht
  refine ⟨ttFirstRankEntryTime S (fun _ => 0)+n,h.label,?_,?_⟩
  · exact h.rank
  · simpa only [ttFirstRankValue,Nat.zero_min] using h.cap
