-- Literal public theorem header from MachineTTChildSVD.lean: ttChild_svd_at
example : ∀ {start : State ttProgram} {trace : Trace ttProgram}
    (ht : LegalTrace ttProgram start trace) (offset : ℕ) (s : State ttProgram)
    (he : trace.state offset=s) (hl : s.label=237) (hs : s.status=.running)
    (L : Layout) (level : ℕ) (hlevel : level<L.order)
    (hrows : s.natural 85≤L.entries) (hcols : s.natural 87≤L.entries)
    (hq : s.natural 86≤s.natural 85)
    (hframe : s.natural 94=L.frame level)
    (hsigma : s.natural 103=L.childSigma) (hvbase : s.natural 104=L.childV)
    (hcurrent : L.workA≤s.natural 89),
  (
    ∃ U : Matrix (Fin (s.natural 85)) (Fin (s.natural 85)) ℝ,
    ∃ sigma : Fin (min (s.natural 85) (s.natural 87)) → ℝ,
    ∃ V : Matrix (Fin (s.natural 87)) (Fin (s.natural 87)) ℝ,
      ValidSVD (readMatrix s.memory (s.natural 85) (s.natural 87) (s.natural 89)) U sigma V ∧
      (trace.state (offset+4)).label=241 ∧ (trace.state (offset+4)).status=.running ∧
      (trace.state (offset+4)).natural 88=s.natural 85 ∧
      (trace.state (offset+4)).natural 108=1 ∧
      (trace.state (offset+4)).memory=
        writeSVD s.memory (L.frame level) L.childSigma L.childV U sigma V ∧
      StoresFrame (trace.state (offset+4)).memory (L.frame level) (s.natural 85)
        (U.submatrix id (Fin.castLE hq)) ∧
      (∀ address,s.natural 89≤address →
        (trace.state (offset+4)).memory address=s.memory address) ∧
      (∀ address,TTStageProtected L level address →
        (trace.state (offset+4)).memory address=s.memory address) ∧
      (∀ j : Fin 128,j≠88 → j≠108 → (trace.state (offset+4)).natural j=s.natural j) ∧
      (trace.state (offset+4)).real=s.real ∧
      CostThrough ttProgram trace (offset+4)≤CostThrough ttProgram trace offset+
        (s.natural 85+s.natural 87+1)^3+3
  ) := @NLA.Proofs.TR04.ttChild_svd_at

-- The actual full child reply and its literal return jointly supply the
-- frame and the recursion's protected-memory/global-state invariant.
example {start : State ttProgram} {trace : Trace ttProgram}
    (ht : LegalTrace ttProgram start trace) (offset : ℕ) (s : State ttProgram)
    (he : trace.state offset=s) (hl : s.label=237) (hs : s.status=.running)
    (L : Layout) (level : ℕ) (hlevel : level<L.order)
    (hrows : s.natural 85≤L.entries) (hcols : s.natural 87≤L.entries)
    (hq : s.natural 86≤s.natural 85) (hframe : s.natural 94=L.frame level)
    (hsigma : s.natural 103=L.childSigma) (hv : s.natural 104=L.childV)
    (hcurrent : L.workA≤s.natural 89) :
    ∃ U : Matrix (Fin (s.natural 85)) (Fin (s.natural 85)) ℝ,
    ∃ sigma : Fin (min (s.natural 85) (s.natural 87)) → ℝ,
    ∃ V : Matrix (Fin (s.natural 87)) (Fin (s.natural 87)) ℝ,
      ValidSVD (readMatrix s.memory (s.natural 85) (s.natural 87) (s.natural 89)) U sigma V ∧
      (trace.state (offset+4)).label=241 ∧
      TensorFormat.StoresFrame (trace.state (offset+4)).memory (L.frame level) (s.natural 85)
        (U.submatrix id (Fin.castLE hq)) ∧
      TTStagePreserves L level s (trace.state (offset+4)) ∧
      CostThrough ttProgram trace (offset+4)≤CostThrough ttProgram trace offset+
        (s.natural 85+s.natural 87+1)^3+3 := by
  obtain ⟨U,sigma,V,hvalid,hlabel,_,_,_,_,hQ,_,hmem,hn,_,hc⟩ :=
    ttChild_svd_at ht offset s he hl hs L level hlevel hrows hcols hq hframe hsigma hv hcurrent
  exact ⟨U,sigma,V,hvalid,hlabel,hQ,ttStagePreserves_of_childSvd hmem hn,hc⟩

-- A complete child stage preserves the parent's actual frame and metadata.
example {L : Layout} {level : ℕ} {s t : State ttProgram}
    (h : TTStagePreserves L (level+1) s t) {m q stride : ℕ}
    (Q : Matrix (Fin m) (Fin q) ℝ) (hq : q≤stride)
    (hm : m≤L.entries) (hstride : stride≤L.entries)
    (hQ : TensorFormat.StoresFrame s.memory (L.frame level) stride Q)
    (values : Fin 4 → ℕ)
    (hv : ∀ f : Fin 4,s.memory (L.metadataCell level f.val)=(values f : ℝ)) :
    TensorFormat.StoresFrame t.memory (L.frame level) stride Q ∧
      ∀ f : Fin 4,t.memory (L.metadataCell level f.val)=(values f : ℝ) := by
  refine ⟨ttStagePreserves_prior_frame h Q hq hm hstride hQ,?_⟩
  intro f
  exact (ttStagePreserves_prior_metadata h f).trans (hv f)
