-- Literal public theorem header from MachineTTInitialFamily.lean: ttInitial_explicit_family
example : ∀ (S : Shape) (r : Ranks S) (A : Tensor S)
    (hr : ∀ j,1≤r j) (trace : Trace ttProgram)
    (ht : LegalTrace ttProgram (Start ttProgram S r A) trace),
  (
    ∃ offset n : ℕ,∃ R : TTFirstReply (trace.state offset),
      offset≤102*(Size S r+1) ∧
      (∃ hi : TTInitialSVDData S r A (trace.state offset),
        TTFirstFamilyPhase R (hi.rank_pos hr) (trace.state (offset+n))) ∧
      n≤9*min ((trace.state offset).natural 81) ((trace.state offset).natural 82)+24 ∧
      TTAllocationRegisters (ofShape S) (trace.state (offset+n)) ∧
      (trace.state (offset+n)).natural 76=0 ∧
      0<(trace.state (offset+n)).natural 75 ∧
      (∀ a,a<(ofShape S).firstU → (trace.state (offset+n)).memory a=InputMemory S r A a) ∧
      readMatrix (trace.state (offset+n)).memory ((trace.state offset).natural 81)
        ((trace.state offset).natural 81) ((trace.state offset).natural 72)=R.U ∧
      (∀ k,(trace.state (offset+n)).memory ((trace.state offset).natural 73+k.val)=R.sigma k) ∧
      readMatrix (trace.state (offset+n)).memory ((trace.state offset).natural 82)
        ((trace.state offset).natural 82) ((trace.state offset).natural 74)=R.V ∧
      offset+n≤1024*(Size S r+1)^3 ∧
      CostThrough ttProgram trace (offset+n)≤1024*(Size S r+1)^3
  ) := @NLA.Proofs.TR04.ttInitial_explicit_family

-- Actual original-input first-family endpoint, without a preloaded header,
-- optimizer, SVD basis or evaluation result.
example (S : Shape) (r : Ranks S) (A : Tensor S) (hr : ∀ j,1≤r j)
    (trace : Trace ttProgram) (ht : LegalTrace ttProgram (Start ttProgram S r A) trace) :
    ∃ time : ℕ,time≤1024*(Size S r+1)^3 ∧
      (trace.state time).label=189 ∧ (trace.state time).status=.running ∧
      TTAllocationRegisters (ofShape S) (trace.state time) ∧
      (trace.state time).natural 76=0 ∧ 0<(trace.state time).natural 75 ∧
      (∀ a,a<(ofShape S).firstU → (trace.state time).memory a=InputMemory S r A a) ∧
      CostThrough ttProgram trace time≤1024*(Size S r+1)^3 := by
  obtain ⟨offset,n,R,_,⟨hi,hphase⟩,_,ha,hj,hcount,hmem,_,_,_,htime,hcost⟩ :=
    ttInitial_explicit_family S r A hr trace ht
  exact ⟨offset+n,htime,hphase.1,hphase.2.1,ha,hj,hcount,hmem,hcost⟩
