      ((tr.state time).natural 72) ((tr.state time).natural 73) ((tr.state time).natural 74)) :
    ∃ U : Matrix (Fin ((tr.state time).natural 81)) (Fin ((tr.state time).natural 81)) ℝ,
    ∃ sigma : Fin (min ((tr.state time).natural 81) ((tr.state time).natural 82)) → ℝ,
    ∃ V : Matrix (Fin ((tr.state time).natural 82)) (Fin ((tr.state time).natural 82)) ℝ,
      ValidSVD (readMatrix (tr.state time).memory ((tr.state time).natural 81)
        ((tr.state time).natural 82) ((tr.state time).natural 92)) U sigma V ∧

        { (tr.state time) with

      tr.cost time=(((tr.state time).natural 81)+((tr.state time).natural 82)+1)^3 := by
  apply trace_svd_at ht time 81 82 92 72 73 74 168 _ hs hd
  rw [hl]
  exact ttProgram_code_first_svd
