    (ht : LegalTrace ttProgram start trace) (offset : ℕ) (s : State ttProgram)
    (he : trace.state offset=s) (hl : s.label=189) (hs : s.status=.running)
    {R : TTFirstReply s0} {hr : 1≤s0.natural 80} {L : Layout}
    (h : TTFirstContext R hr L s) (hj : s.natural 76<s.natural 75),
  (
    let time := s0.natural 81*(11*s.natural 106+7)+24
    let t := trace.state (offset+time)
    ∃ j : Fin R.family.count,j.val=s.natural 76 ∧ s.natural 106=R.family.width ∧
      time≤24*(L.entries+1)^2 ∧ t.label=241 ∧ t.status=.running ∧
      StoresFrame t.memory L.frames R.family.width (R.family.frame j) ∧
      TTFirstContext R hr L t ∧ TTStagePreserves L 0 s t ∧
      TTCandidateMetadataFields s t ∧
      CostThrough ttProgram trace (offset+time)≤CostThrough ttProgram trace offset+24*(L.entries+1)^2
  ) := @NLA.Proofs.TR04.TTFirstContext.gather
end

-- The actual first-family endpoint has at most the original dense input size.
example {S : Shape} {r : Ranks S} {A : Tensor S} {s0 t : State ttProgram}
    (hi : TTInitialSVDData S r A s0) {R : TTFirstReply s0} {hr : 1≤s0.natural 80}
    (h : TTFirstFamilyPhase R hr t) (ha : TTAllocationRegisters (ofShape S) t) :
    t.natural 75≤DenseSize S := (h.context hi ha).count_le
-- Every actual child stage retains the SAME saved first family.
example {s0 s t : State ttProgram} {R : TTFirstReply s0} {hr : 1≤s0.natural 80}
    {L : Layout} (h : TTFirstContext R hr L s) (level : ℕ)
    (hp : TTStagePreserves L level s t) :
    TTFirstContext R hr L t ∧ t.natural 75=R.family.count :=
  ⟨h.of_stage hp,(h.of_stage hp).count_eq⟩
