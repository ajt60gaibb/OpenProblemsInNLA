
namespace NLA.Proofs.TR04
open NLA.Computation.SVDMachine NLA.Computation.TensorTrainApproximation TTMemoryLayout

-- Original-input composition uses no caller-supplied allocation, solver result
-- or decoded rank. The actual candidate0 is in the computed nonempty family.
example (S : Shape) (r : Ranks S) (A : Tensor S) (hr : ∀ j,1≤r j)
    (trace : Trace ttProgram) (ht : LegalTrace ttProgram (Start ttProgram S r A) trace) :
    ∃ t : ℕ,t≤1024*(Size S r+1)^3 ∧
      (trace.state t).label=189 ∧ (trace.state t).status=.running ∧
      (trace.state t).natural 0=S.order ∧
      (trace.state t).natural 76=0 ∧ (trace.state t).natural 76<(trace.state t).natural 75 ∧
      (∀ a,a<(ofShape S).firstU → (trace.state t).memory a=InputMemory S r A a) ∧
      CostThrough ttProgram trace t≤1024*(Size S r+1)^3 := by
  obtain ⟨offset,n,R,_,⟨hi,hphase⟩,_,_,hj,hcount,hprefix,_,_,_,htime,hcost⟩ :=
    ttInitial_explicit_family S r A hr trace ht
  refine ⟨offset+n,htime,hphase.1,hphase.2.1,ttProgram_trace_order S r A trace ht _,hj,?_,hprefix,hcost⟩
  omega

-- A prior frame becomes protected after its level and stays unchanged under
-- every later child full-SVD write, for arbitrary legal response arrays.
example (L : Layout) (earlier level address : ℕ) (hprev : earlier<level) (hl : level<L.order)
    (ha : L.frame earlier≤address ∧ address<L.frame earlier+L.entries^2)
    {m l : ℕ} (hm : m≤L.entries) (hc : l≤L.entries) (memory : ℕ → ℝ)
    (U : Matrix (Fin m) (Fin m) ℝ) (sigma : Fin (min m l) → ℝ) (V : Matrix (Fin l) (Fin l) ℝ) :
    writeSVD memory (L.frame level) L.childSigma L.childV U sigma V address=memory address := by
  apply ttStageProtected_child_svd L level hl hm hc memory U sigma V address
  exact ttStageProtected_mono L (by omega) (ttStageProtected_own_frame L earlier address ha)

end NLA.Proofs.TR04
