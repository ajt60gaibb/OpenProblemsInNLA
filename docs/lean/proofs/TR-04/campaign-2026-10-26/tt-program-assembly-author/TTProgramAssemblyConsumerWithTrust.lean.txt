example (S : Shape) (r : Ranks S) (A : Tensor S) (tr : Trace ttProgram)
    (ht : LegalTrace ttProgram (Start ttProgram S r A) tr) :
    (tr.state (headerProductReturnTime (shapeHeaderValues S) S.order)).label=10 ∧
    (tr.state (headerProductReturnTime (shapeHeaderValues S) S.order)).status=.running ∧
    (tr.state (headerProductReturnTime (shapeHeaderValues S) S.order)).natural 2=DenseSize S ∧
    (tr.state (headerProductReturnTime (shapeHeaderValues S) S.order)).memory=InputMemory S r A ∧
    CostThrough ttProgram tr (headerProductReturnTime (shapeHeaderValues S) S.order)≤7*Size S r :=
  ttProgram_header_result S r A tr ht

example (S : Shape) (r : Ranks S) (A : Tensor S) (tr : Trace ttProgram)
    (ht : LegalTrace ttProgram (Start ttProgram S r A) tr) :
    (tr.state (headerProductReturnTime (shapeHeaderValues S) S.order)).natural 97=0 ∧
    (tr.state (headerProductReturnTime (shapeHeaderValues S) S.order)).natural 127=0 :=
  ⟨ttProgram_header_natural_outside S r A tr ht 97 (by decide),
   ttProgram_header_natural_outside S r A tr ht 127 (by decide)⟩

    ttProgram.code label=ttControllerProgram.code label := ttProgram_code_caller label hl

example (start : State ttProgram) (tr : Trace ttProgram) (ht : LegalTrace ttProgram start tr)
    (time : ℕ) (hl : (tr.state time).label=167) (hs : (tr.state time).status=.running)
    (hd : OutputBlocksDisjoint ((tr.state time).natural 81) ((tr.state time).natural 82)
      ((tr.state time).natural 72) ((tr.state time).natural 73) ((tr.state time).natural 74)) :
    ∃ U : Matrix (Fin ((tr.state time).natural 81)) (Fin ((tr.state time).natural 81)) ℝ,
    ∃ sigma : Fin (min ((tr.state time).natural 81) ((tr.state time).natural 82)) → ℝ,
    ∃ V : Matrix (Fin ((tr.state time).natural 82)) (Fin ((tr.state time).natural 82)) ℝ,
      ValidSVD (readMatrix (tr.state time).memory ((tr.state time).natural 81)
        ((tr.state time).natural 82) ((tr.state time).natural 92)) U sigma V ∧
      tr.state (time+1) =
        { (tr.state time) with
          label := 168
          memory := writeSVD (tr.state time).memory ((tr.state time).natural 72)
            ((tr.state time).natural 73) ((tr.state time).natural 74) U sigma V } ∧
      tr.cost time=(((tr.state time).natural 81)+((tr.state time).natural 82)+1)^3 := by
  apply trace_svd_at ht time 81 82 92 72 73 74 168 _ hs hd
  rw [hl]
  exact ttProgram_code_first_svd
