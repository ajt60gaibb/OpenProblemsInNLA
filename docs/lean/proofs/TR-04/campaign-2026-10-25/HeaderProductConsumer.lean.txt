example (memory : ℕ → ℝ) (d : ℕ) (values : ℕ → ℕ)
    (hm : ∀ j<d,memory j = (values j : ℝ)) (tr : Trace headerProductProgram)
    (ht : LegalTrace headerProductProgram (initial headerProductProgram d memory) tr) :
    (tr.state (headerPrefixTime values d+4)).status = .halted (∏ i ∈ Finset.range d,values i) ∧
    (tr.state (headerPrefixTime values d+4)).memory = memory ∧
    CostThrough headerProductProgram tr (headerPrefixTime values d+4) ≤ headerPrefixTime values d+4 := by
  obtain ⟨counter,reals,hv⟩ := headerProduct_trace_result memory d values hm tr ht
  change tr.state (headerPrefixTime values d+4) = _ at hv
  refine ⟨?_,?_,headerProduct_trace_cost memory d values tr ht⟩
  · rw [hv];rfl
  · rw [hv];rfl
example (memory : ℕ → ℝ) (tr : Trace headerProductProgram)
    (ht : LegalTrace headerProductProgram (initial headerProductProgram 0 memory) tr) :
    (tr.state 4).status = .halted 1 ∧ (tr.state 4).memory = memory := by
  obtain ⟨counter,reals,hv⟩ := headerProduct_trace_result memory 0 (fun _ => 0)
    (by intro j hj;omega) tr ht
  have heq : headerProductTime (fun _ => 0) 0 = 4 := rfl
  rw [heq] at hv
  rw [hv]
  exact ⟨rfl,rfl⟩
example (Q : Program) (E : RoutineEmbedding headerProductProgram Q)
    (memory : ℕ → ℝ) (d : ℕ) (values : ℕ → ℕ)
    (hm : ∀ j<d,memory j = (values j : ℝ)) (start : State Q) (tr : Trace Q)
    (ht : LegalTrace Q start tr) (offset : ℕ)
    (hs : E.Relates (initial headerProductProgram d memory) (tr.state offset)) :
    ∃ counter : ℕ,∃ reals : Fin 2 → ℝ,
      E.Relates (headerProductReturnState memory d values counter reals)
        (tr.state (offset+headerPrefixTime values d+3)) ∧
      CostThrough Q tr (offset+headerPrefixTime values d+3) ≤
        CostThrough Q tr offset+headerPrefixTime values d+3 := by
  simpa only [headerProductReturnTime,Nat.add_assoc] using E.headerProduct_run_at memory d values hm ht offset hs
example (memory : ℕ → ℝ) (d : ℕ) (values : ℕ → ℕ)
    (hm : ∀ j<d,memory j = (values j : ℝ)) (k : ℕ) (hk : k<headerProductReturnTime values d) :
    instructionIsHalt (headerProductProgram.code ((ordinaryStep headerProductProgram)^[k]
      (initial headerProductProgram d memory)).label) = false :=
  ordinary_nohalt_prefix_of_running _ _ (headerProduct_return_running memory d values hm) k hk
