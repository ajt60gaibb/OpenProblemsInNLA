import Solution
import LeanCert.Tactic.Verification
set_option autoImplicit false
set_option leancert.trust "kernel"

open Lean Elab Command in
elab "#assert_proof_constant " n:ident : command => do
  let decl ← liftCoreM <| realizeGlobalConstNoOverloadWithInfo n
  match (← getEnv).find? decl with
  | some (.thmInfo _) => pure ()
  | _ => throwErrorAt n "expected a proved theorem constant"

#assert_proof_constant NLA.Proofs.TR26.nonisotropic_degree
#assert_trust kernel NLA.Proofs.TR26.nonisotropic_degree
#print axioms NLA.Proofs.TR26.nonisotropic_degree
#assert_proof_constant NLA.Proofs.TR26.isotropic_degree
#assert_trust kernel NLA.Proofs.TR26.isotropic_degree
#print axioms NLA.Proofs.TR26.isotropic_degree
#assert_proof_constant NLA.Proofs.TR26.target
#assert_trust kernel NLA.Proofs.TR26.target
#print axioms NLA.Proofs.TR26.target
