    MvPolynomial.eval c (fixedDiscriminant k) = 0 ↔
      ∃ a b : ℂ, (a ≠ 0 ∨ b ≠ 0) ∧

    (L : SymIndex d → Polynomial ℂ) (P : Polynomial ℂ) (hP : P ≠ 0)
    (hcritical : ∀ s (A : Parameter d), A ≠ 0 → P.eval s ≠ 0 →
      linearCriticalValue L s A = 0 → OpenCritical (ζ s) A)
    (t : ℂ) (H : Parameter d) (hH : H ≠ 0) (hcurve : OnCurve (ζ t))
    (j : SymIndex d) (hj : (L j).eval t ≠ 0)
    (ht : linearCriticalValue L t H = 0)
    (hderiv : (linearCriticalPolynomial L H).derivative.eval t = 0)
    (v : CurveVector d) (hv : ∀ i, HasDerivAt (fun s => ζ s i) (v i) t)
    (hind : LinearIndependent ℂ ![ζ t,v]) : Ramification d (ζ t) H :=
  chart_ramification_of_double_root d ζ hζ L P hP hcritical t H hH hcurve j hj
    ht hderiv v hv hind

-- A quadratic a² has a repeated root at the projective infinity (a,b)=(0,1).
example : MvPolynomial.eval (fun i : Fin 3 => if i.val = 0 then (1 : ℂ) else 0)
    (fixedDiscriminant 2) = 0 := by
  apply multiple_projective_root_vanishes (by omega)
  refine ⟨0,1,Or.inr one_ne_zero,?_,?_⟩
  all_goals norm_num [partialACoeff,partialBCoeff,lowerIndex,upperIndex,Fin.sum_univ_two]

-- The fixed quadratic a*b has two simple projective roots, one at infinity.
example : MvPolynomial.eval (fun i : Fin 3 => if i.val = 1 then (1 : ℂ) else 0)
    (fixedDiscriminant 2) ≠ 0 := by
  rw [eval_fixedDiscriminant_eq_zero_iff (by omega)]
  rintro ⟨a,b,hab,ha,hb⟩
  norm_num [partialACoeff,partialBCoeff,lowerIndex,upperIndex,Fin.sum_univ_two] at ha hb
  rcases hab with ha0 | hb0
  · exact ha0 hb
  · exact hb0 ha
