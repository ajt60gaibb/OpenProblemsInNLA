    squaredFrobenius (B*gaussianBlock n q offset ω) 0 =
      gaussianQuadratic (fun p : Fin n × Fin q => (residualSingularValue B p.1)^2)
        (residualGaussianCoordinates B q offset ω) := residualGaussian_energy B q offset ω

theorem literal_lower_unscaled {n : ℕ} (B : Matrix (Fin n) (Fin n) ℝ)
    (q offset r : ℕ) (hr : r<n) (η : ℝ) (hη0 : 0<η) (hη1 : η≤1) :
    GaussianLaw.real {ω | squaredFrobenius (B*gaussianBlock n q offset ω) 0 <
      (q : ℝ)*(1-η)*residualTailEnergy B r} ≤ Real.exp (-(q : ℝ)*(r : ℝ)*η/4) := residualGaussian_lower_tail B q offset r hr η hη0 hη1
theorem literal_upper_unscaled {n : ℕ} (B : Matrix (Fin n) (Fin n) ℝ)
    (q offset : ℕ) (η : ℝ) (hη0 : 0<η) (hη1 : η≤1) :
    GaussianLaw.real {ω | (q : ℝ)*(1+η)*squaredFrobenius B 0 <
      squaredFrobenius (B*gaussianBlock n q offset ω) 0} ≤ Real.exp (-(q : ℝ)*η^2/8) := residualGaussian_upper_tail B q offset η hη0 hη1
theorem literal_actual_warm {n m : ℕ} (F : Family n m) (A : Matrix (Fin n) (Fin n) ℝ)
    (q offset r : ℕ) (hq : 0<q) (hr : r<n) (η : ℝ) (hη0 : 0<η) (hη1 : η<1) :
    GaussianLaw.real {ω | ((1+η)/(1-η))*(optimum A F)^2 <
      residualTailEnergy (A-F (gaussianWarmIndex F A q offset ω)) r} ≤
      (m+2 : ℝ)*Real.exp (-(q : ℝ)*(r : ℝ)*η/4)+Real.exp (-(q : ℝ)*η^2/8) := gaussianWarmIndex_tail_bound F A q offset r hq hr η hη0 hη1
theorem literal_program_warm {n m : ℕ} (F : Family n m) (A : Matrix (Fin n) (Fin n) ℝ)
    (ε : ℝ) (hsketch : firstWidth m ε+secondWidth m ε+thirdWidth m ε<n) (ω : ℕ→ℝ) :
    gaussianWarmIndex F A (firstWidth m ε) 0 ω = sketchWarmIndex F A (makePlan n m ε ω) := gaussianWarmIndex_eq_sketchWarmIndex F A ε hsketch ω
theorem literal_program_warm_tail {n m : ℕ} (F : Family n m) (A : Matrix (Fin n) (Fin n) ℝ)
    (ε : ℝ) (hsketch : firstWidth m ε+secondWidth m ε+thirdWidth m ε<n)
