import Solution
import Reviewed.RE06
import LeanCert.Tactic.Verification
set_option autoImplicit false
set_option leancert.trust "kernel"
open Lean Elab Command in
elab "#inventory_theorem " n:ident : command => do
  let d ← liftCoreM <| realizeGlobalConstNoOverloadWithInfo n
  match (← getEnv).find? d with
  | some (.thmInfo _) => pure ()
  | _ => throwErrorAt n "inventory expects genuine theorem"
namespace Inventory.RE06.Package

theorem consumeSVD : NLA.Computation.NonadaptiveMatrixQuery.SVDTotal :=
  NLA.Proofs.RE06.svd_total

theorem consumeQueries (n m : ℕ) (hn : 0 < n) (ε : ℝ)
    (hε0 : 0 < ε) (hεh : ε < 1 / 2) :
    NLA.Computation.NonadaptiveMatrixQuery.QueryBound n m ε 4000000 0 :=
  NLA.Proofs.RE06.query_bound n m hn ε hε0 hεh

theorem consumeSuccess (n m : ℕ) (hn : 0 < n)
    (F : NLA.Computation.NonadaptiveMatrixQuery.Family n m) (hF : Function.Injective F)
    (A : Matrix (Fin n) (Fin n) ℝ) (ε : ℝ) (hε0 : 0 < ε) (hεh : ε < 1 / 2) :
    NLA.Computation.NonadaptiveMatrixQuery.Successful F A ε :=
  NLA.Proofs.RE06.successful n m hn F hF A ε hε0 hεh

theorem consumeOriginal : NLA.Statements.RE06.OriginalClaim :=
  NLA.Proofs.RE06.original_claim

theorem consumeExplicit : NLA.Statements.RE06.ExplicitClaim :=
  NLA.Proofs.RE06.explicit_claim

theorem consumeTarget : NLA.Statements.RE06.Target := NLA.Proofs.RE06.target

theorem liveReviewedIdentity :
    NLA.Statements.RE06.Target = NLA.ReviewedStatements.RE06.Target := rfl

theorem consumeReviewedTarget : NLA.ReviewedStatements.RE06.Target := by
  rw [← liveReviewedIdentity]
  exact NLA.Proofs.RE06.target
end Inventory.RE06.Package
