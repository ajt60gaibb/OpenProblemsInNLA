 theorem ActualPythagoras {n k c ell : ℕ} (Y : Matrix (Fin n) (Fin k) ℝ)
     (C : Matrix (Fin n) (Fin c) ℝ) (Ht : Matrix (Fin ell) (Fin n) ℝ) :
     squaredFrobenius (C-Y*pseudoInverse (Ht*Y)*(Ht*C)) 0 =
       squaredFrobenius (actualRangeResidual Y C) 0+actualRegressionEnergy Y C Ht :=
   actual_repair_pythagoras Y C Ht
 theorem ActualSVDBound {n k ell : ℕ} (C : Matrix (Fin n) (Fin n) ℝ)
     (r : ℕ) (hr : r ≤ n) (G : Matrix (Fin n) (Fin k) ℝ) (Ht : Matrix (Fin ell) (Fin n) ℝ)
     (hOmega : headCoordinates C r hr G*pseudoInverse (headCoordinates C r hr G)=1) :
     squaredFrobenius (C-(C*G)*pseudoInverse (Ht*(C*G))*(Ht*C)) 0 ≤
       residualTailEnergy C r+actualExcessMajorant C r hr G Ht := actual_svd_repair_majorant C r hr G Ht hOmega
 theorem ActualMeasurableExcess {Ω : Type*} [MeasurableSpace Ω] {n k ell : ℕ}
     (C : Matrix (Fin n) (Fin n) ℝ) (r : ℕ) (hr : r ≤ n)
     {G : Ω → Matrix (Fin n) (Fin k) ℝ} {Ht : Ω → Matrix (Fin ell) (Fin n) ℝ}
     (hG : Measurable G) (hHt : Measurable Ht) :
