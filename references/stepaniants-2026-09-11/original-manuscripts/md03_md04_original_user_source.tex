\section*{Source and logical status}

The problems are the matrix statements in
\cite{MD03,MD04}. The cited mathematical input is Guo, Fang, and Lu,
\emph{Vector Balancing via Directional Total Variation}
\cite{GFL26}, specifically \textbf{Theorem~1.1, p.~1}, and
\textbf{Corollary~1.2, p.~2}. Both give the requested estimates with
\[
  \boxed{\Cstar=3\sqrt{2\pi}.}
\]

\textbf{Qualification.}
Reference~\cite{GFL26} is the preprint version submitted on
10 September 2026. The conclusions below are conditional on the correctness
of its cited results. This document supplies the complete applications of
those results to the two problem statements; it is not an independent
verification of the preprint's proof or a claim of a refereed resolution.
The version and theorem locations were checked on 11 September 2026.

For a real matrix $A\in\R^{m\times n}$, write
\[
 \disc(A):=\min_{\varepsilon\in\{-1,1\}^{n}}
             \|A\varepsilon\|_{\infty},
 \qquad
 \|y\|_{\infty}:=\max_{1\leq i\leq m}|y_i|.
\]
The minimum is attained because its domain is a finite, nonempty set.

\section{MD-03: the Koml\'os discrepancy statement}

\begin{theorem}[MD-03, using {\cite[Theorem~1.1]{GFL26}}]
Let $m,n\geq1$ be integers and let $A=(a_{ij})\in\R^{m\times n}$
satisfy
\begin{equation}\label{eq:komlos-hyp}
  \sum_{i=1}^{m}a_{ij}^{2}\leq1
  \qquad (1\leq j\leq n).
\end{equation}
Then, subject to the source qualification above, there exists
$\varepsilon\in\{-1,1\}^{n}$ such that
\begin{equation}\label{eq:komlos-conclusion}
  \|A\varepsilon\|_{\infty}<\Cstar.
\end{equation}
In particular, the constant requested in MD-03 can be taken to be
$C=\Cstar$, independently of $m,n$ and $A$.
\end{theorem}

\begin{proof}[Proof by citation]
Let $v_j=A_{\cdot j}\in\R^m$. Condition~\eqref{eq:komlos-hyp} is exactly
$\|v_j\|_2\leq1$. Apply \cite[Theorem~1.1, p.~1]{GFL26} to these $n$
vectors. Its signed sum is $\sum_j\varepsilon_jv_j=A\varepsilon$, so its
conclusion is \eqref{eq:komlos-conclusion}. The strict estimate implies
the non-strict estimate asked for in \cite{MD03}.
\end{proof}

\section{MD-04: the Beck--Fiala discrepancy statement}

\begin{theorem}[MD-04, using {\cite[Corollary~1.2]{GFL26}}]
Let $m,n\geq1$ be integers, let $t$ be an integer with $1\leq t\leq m$,
and let $A=(a_{ij})\in\{0,1\}^{m\times n}$ satisfy
\begin{equation}\label{eq:bf-hyp}
  \sum_{i=1}^{m}a_{ij}\leq t
  \qquad (1\leq j\leq n).
\end{equation}
Then, subject to the source qualification above, there exists
$\varepsilon\in\{-1,1\}^{n}$ such that
\begin{equation}\label{eq:bf-conclusion}
  \|A\varepsilon\|_{\infty}<\Cstar\sqrt{t}.
\end{equation}
Thus $C=\Cstar$ also meets the requirement of MD-04, independently of
$m,n,t$ and $A$.
\end{theorem}

\begin{proof}[Proof by citation]
For a $0$--$1$ column, its sum equals its number of nonzero entries.
Consequently \eqref{eq:bf-hyp} is precisely the sparsity hypothesis of
\cite[Corollary~1.2, p.~2]{GFL26}. That result gives
$\disc(A)<3\sqrt{2\pi t}=\Cstar\sqrt t$.
A signing attaining the defining minimum proves
\eqref{eq:bf-conclusion}, and hence the non-strict inequality in
\cite{MD04}.
\end{proof}

\paragraph{Attribution and scope.}
Both estimates are results asserted and proved in \cite{GFL26}; neither
is a new result of this note. The Beck--Fiala estimate is already a
stated corollary there, not a new corollary claimed here. The signings
in the two statements assign a sign to every column. The problems
\cite{MD03,MD04} ask for existence, so no algorithmic assertion is needed.
